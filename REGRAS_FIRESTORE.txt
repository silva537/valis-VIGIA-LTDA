rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /{document=**} {
      allow read, write: if false;
    }
    match /valis/{docId} {
      allow read, write: if request.auth != null
        && docId in [
          'db_v42',
          'aviso',
          'manutencao',
          'plantoes_backup',
          'dev_log',
          'cadastros_pendentes',
          'solicitacoes',
          'config_app'
        ];
    }
  }
}
